
LIBRARY ieee;
USE ieee.std_logic_1164.ALL;
USE ieee.numeric_std.ALL;
 
-- 4-bit ALU (ULA) controlled by a single push button.
-- The user walks through a sequence of steps, pressing the button once per step:
--   1) choose the operation code      (keys -> "operation")
--   2) enter the first operand        (keys -> "entry1")
--   3) enter the second operand       (keys -> "entry2", binary operations only)
--   4) the result is computed and shown on the LEDs, along with the flags
--   5) one more press clears everything and returns to step 1
--
-- Operation codes (selected with the keys):
--   0000 Addition          0100 XOR
--   0001 Subtraction       0101 XNOR
--   0010 Bit shift left    0110 2's complement (unary)
--   0011 Comparison        0111 Absolute value (unary)
--   any other code -> result "1111"
ENTITY ula IS
    PORT(
        input          : IN STD_LOGIC_VECTOR(3 DOWNTO 0); -- keys: operation code or operand, depending on the step
        clock          : IN STD_LOGIC; -- oscillator E12 (board clock)
        button         : IN STD_LOGIC; -- button used to advance the ULA to the next step
        output         : OUT STD_LOGIC_VECTOR(3 DOWNTO 0) := "0000"; -- leds: 4-bit result
        flag_zero      : OUT STD_LOGIC := '1'; -- led w21: result is zero
        flag_negative  : OUT STD_LOGIC := '0'; -- led y22: result is negative (MSB = '1')
        flag_carryout  : OUT STD_LOGIC := '0'; -- led v20: carry out (or borrow out in a subtraction)
        flag_overflow  : OUT STD_LOGIC := '0' -- led v19: signed overflow
    );
END ula;
 
ARCHITECTURE main OF ula IS
 
    -- Output of the debouncer: a clean, one-clock-cycle pulse for each button press
    -- (the raw "button" signal is noisy because of the mechanical bounce)
    SIGNAL debounced_button : STD_LOGIC;
 
    -- Debouncer component (implemented in a separate file, debounce.vhd)
    COMPONENT debounce
        PORT(
            clk    : IN STD_LOGIC;
            btn_in : IN STD_LOGIC;
            btn_out : OUT STD_LOGIC
        );
    END COMPONENT;
 
BEGIN
 
    -- Connects the raw button to the debouncer; the ULA only uses the filtered pulse
    debounce_instance : debounce
        PORT MAP (
            clk    => clock,
            btn_in => button,
            btn_out => debounced_button
        );
 
-- Main process: a state machine that runs on the board clock.
-- Variables keep their values between clock cycles, so they work as the ULA's memory.
PROCESS(clock)
 
-- Variable to control the state of the ULA
--   "000" -> waiting for the operation code
--   "001" -> waiting for the first operand
--   "010" -> waiting for the second operand (binary operations only)
--   "011" -> compute and show the result
--   "100" -> (any other value) waiting for a press to reset everything
VARIABLE state : STD_LOGIC_VECTOR(2 DOWNTO 0) := "000";
 
-- Variables to store the operation and the entries
VARIABLE operation  : STD_LOGIC_VECTOR(3 DOWNTO 0); -- operation code chosen in the first step
VARIABLE entry1     : STD_LOGIC_VECTOR(3 DOWNTO 0); -- first operand (A)
VARIABLE entry2     : STD_LOGIC_VECTOR(3 DOWNTO 0); -- second operand (B)
 
-- Variable to hold the result temporarily
-- (it is only copied to the "output" LEDs after the operation is finished)
VARIABLE result_control : STD_LOGIC_VECTOR(3 DOWNTO 0) := "0000";
 
-- Variable used to control the carry
-- carry_control(i) is the carry (or borrow) generated by bit i,
-- so carry_control(3) is the carry out of the whole 4-bit operation
VARIABLE carry_control : STD_LOGIC_VECTOR(3 DOWNTO 0) := "0000";
 
BEGIN
 
    -- Everything happens on the rising edge of the clock (synchronous design)
    IF RISING_EDGE(clock) THEN
        -- If the button was pressed (the debouncer produced a pulse), advance one step.
        -- Without a press, nothing changes.
        IF(debounced_button = '1') THEN
        CASE state IS
 
            -- STEP 1: read the operation code
            WHEN "000" =>
                -- The flag LEDs are also used here as a visual indicator of the current step
                -- (zero LED off, negative LED on = waiting for the first operand)
                flag_zero <= '0';
                flag_negative <= '1';
                -- Hold the value of the operation and goes to the next stage
                operation := input;
                state := "001";
 
            -- STEP 2: read the first operand
            WHEN "001" =>
                -- Turns off the indicator LED of the previous step
                flag_negative <= '0';
                -- Hold the value of the entry used
                entry1 := input;
                
                -- Direct the ULA to the next stage depending on the operation chosen
                IF (operation = "0110" OR operation = "0111") THEN
                    -- Unary operations (2's complement and absolute value) use only one operand,
                    -- so the ULA skips the second entry and goes straight to the calculation
                    state := "011"; -- Unary operations
                ELSE
                    -- Binary operations need a second operand
                    state := "010"; -- Binary operations
                    -- Carry-out LED on = waiting for the second operand
                    flag_carryout <= '1';
                END IF;
 
            -- STEP 3: read the second operand (binary operations only)
            WHEN "010" =>
                -- Turns off the indicator LED of the previous step
                flag_carryout <= '0';
                -- Overflow LED on = ready to compute on the next press
                flag_overflow <= '1';
                -- Hold the value of the second entry used
                entry2 := input;
                state := "011";
 
            -- STEP 4: execute the chosen operation and display the result
            WHEN "011" =>
                -- Turns off the indicator LED of the previous step
                flag_overflow <= '0';
                -- Run operations
                CASE operation IS
                    WHEN "0000" => -- Addition
                        -- Signed (2's complement) addition of the two operands
                        result_control := STD_LOGIC_VECTOR(SIGNED(entry1) + SIGNED(entry2));
 
                        -- Carry during the sum
                        -- Each carry follows the full-adder rule: carry = (a AND b) OR (carry_in AND a) OR (carry_in AND b)
                        -- Bit 0 has no carry in, so its carry is just (a AND b)
                        carry_control(0) := entry1(0) AND entry2(0);
                        carry_control(1) := (carry_control(0) AND entry1(1)) OR (carry_control(0) AND entry2(1)) OR (entry1(1) AND entry2(1));
                        carry_control(2) := (carry_control(1) AND entry1(2)) OR (carry_control(1) AND entry2(2)) OR (entry1(2) AND entry2(2));
                        carry_control(3) := (carry_control(2) AND entry1(3)) OR (carry_control(2) AND entry2(3)) OR (entry1(3) AND entry2(3));
 
                        -- Detecting carryout and overflow
                        -- Carry out = carry generated by the MSB
                        -- Signed overflow = carry into the MSB XOR carry out of the MSB
                        flag_carryout <= carry_control(3);
                        flag_overflow <= carry_control(3) XOR carry_control(2);
 
                    WHEN "0001" => -- Subtraction
                        -- Signed (2's complement) subtraction: entry1 - entry2
                        result_control := STD_LOGIC_VECTOR(SIGNED(entry1) - SIGNED(entry2));
 
                        -- Carry during the subtraction (borrow out)
                        -- Bit i generates a borrow when entry1(i) < entry2(i), or when the bits are equal
                        -- and a borrow came from the previous bit
                        -- (NOT (a XOR b) is '1' when the two bits are equal)
                        carry_control(0) := (NOT entry1(0)) AND entry2(0);
                        carry_control(1) := (carry_control(0) AND (NOT (entry1(1) XOR entry2(1)))) OR ((NOT entry1(1)) AND entry2(1));
                        carry_control(2) := (carry_control(1) AND (NOT (entry1(2) XOR entry2(2)))) OR ((NOT entry1(2)) AND entry2(2));
                        carry_control(3) := (carry_control(2) AND (NOT (entry1(3) XOR entry2(3)))) OR ((NOT entry1(3)) AND entry2(3));
 
                        -- Detecting carryout and overflow
                        -- Same rule as the addition: borrow out of the MSB and borrow into the MSB
                        flag_carryout <= carry_control(3);
                        flag_overflow <= carry_control(3) XOR carry_control(2);
 
                    WHEN "0010" => -- Bit Shift
                        -- Shifts entry1 to the left by the number of positions given by entry2
                        -- (entry2 is read as an unsigned number; vacated bits are filled with zeros)
                        result_control := STD_LOGIC_VECTOR(SHIFT_LEFT(UNSIGNED(entry1), TO_INTEGER(UNSIGNED(entry2))));
 
                    WHEN "0011" => -- Comparison
                        -- Stores the subtraction at the variable that will then be overwritten
                        -- (the sign and the zero test of A - B tell how A and B compare)
                        result_control := STD_LOGIC_VECTOR(SIGNED(entry1) - SIGNED(entry2));
 
                        -- The result is a one-hot code shown on the LEDs:
                        --   "0001" -> A < B,  "0010" -> A = B,  "0100" -> A > B
                        IF result_control = "0000" THEN -- A = B
                            result_control := "0010";
 
                        ELSIF result_control(3) = '1' THEN -- A < B
                            result_control := "0001";
 
                        ELSE -- A > B
                            result_control := "0100";
 
                        END IF;
 
                    WHEN "0100" => -- XOR
                        -- Bitwise XOR between the operands
                        result_control := entry1 XOR entry2;
 
                    WHEN "0101" => -- XNOR
                        -- Bitwise XNOR between the operands
                        result_control := entry1 XNOR entry2;
 
                    WHEN "0110" => -- 2's Complement
                        -- Unary operation: invert all bits of entry1 and add 1
                        result_control := STD_LOGIC_VECTOR(UNSIGNED(NOT entry1) + 1);
 
                    WHEN "0111" => -- Absolute Value
                        -- Unary operation: if entry1 is negative (MSB = '1'), take its 2's complement;
                        -- otherwise it is already positive and is kept as it is
                        IF entry1(3) = '1' THEN
                            result_control := STD_LOGIC_VECTOR(UNSIGNED(NOT entry1) + 1);
                        ELSE
                            result_control := entry1;
                        END IF;
 
                    WHEN OTHERS =>
 
                        -- Generic result for any other type of operation chosen
                        result_control := "1111";
 
                END CASE;
 
                -- Update the result with the temporary result
                -- (the LEDs only change here, after the whole calculation is done)
                output <= result_control;
 
                -- Check the flags for negative and zero!
                -- Zero: all bits of the result are '0'
                -- Negative: MSB of the result is '1' (2's complement representation)
                IF result_control = "0000" THEN
                    flag_zero <= '1';
                ELSIF result_control(3) = '1' THEN
                    flag_negative <= '1';    
                END IF;
            
                -- Send to the state where things are resetted        
                state := "100";
 
            -- STEP 5 (state "100" or any invalid state): reset everything
            WHEN OTHERS =>
 
                -- Clears the flags
                -- (flag_zero goes back to '1' because the cleared output is zero)
                flag_zero     <= '1';
                flag_negative <= '0';
                flag_overflow <= '0';
                flag_carryout <= '0';
                -- Resets the result and the state
                output <= "0000";
                state := "000";
 
        END CASE;
        END IF; -- end of the button check
    END IF; -- end of the clock edge check
 
-- End of the main process (clocked state machine)
END PROCESS;
 
-- End of the ULA architecture
END main;
